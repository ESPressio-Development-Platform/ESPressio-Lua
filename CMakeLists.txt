cmake_minimum_required(VERSION 3.16)
project(ESPressioLua LANGUAGES CXX)
set(ESPRESSIO_SYSTEM_DIR "" CACHE PATH "ESPressio-System checkout at primitives_redesign")
set(ESPRESSIO_PRIMITIVE_DIR "" CACHE PATH "Optional ESPressio-Primitive checkout at primitives_redesign for Primitive discovery tests")
if(NOT EXISTS "${ESPRESSIO_SYSTEM_DIR}/src/ESPressio_Memory.hpp")
    message(FATAL_ERROR "Set ESPRESSIO_SYSTEM_DIR to an ESPressio-System primitives_redesign checkout")
endif()
add_library(espressio_lua STATIC src/ESPressio_LuaRuntime.cpp)
target_include_directories(espressio_lua PUBLIC src "${ESPRESSIO_SYSTEM_DIR}/src")
target_compile_features(espressio_lua PUBLIC cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(espressio_lua PUBLIC -fexceptions)
endif()
option(ESPRESSIO_LUA_BUILD_TESTS "Build native binding regression tests" ON)
option(ESPRESSIO_LUA_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
if(ESPRESSIO_LUA_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(espressio_lua PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(espressio_lua PUBLIC -fsanitize=address,undefined)
endif()
if(ESPRESSIO_LUA_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(binding_tests tests/binding_tests.cpp)
    target_link_libraries(binding_tests PRIVATE espressio_lua Threads::Threads)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(binding_tests PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME binding_tests COMMAND binding_tests)
    add_executable(demo_test tests/demo_test.cpp)
    target_link_libraries(demo_test PRIVATE espressio_lua)
    add_test(NAME demo_test COMMAND demo_test)

    if(NOT EXISTS "${ESPRESSIO_PRIMITIVE_DIR}/src/ESPressio_TypeDirectory.hpp")
        message(FATAL_ERROR "Set ESPRESSIO_PRIMITIVE_DIR to an ESPressio-Primitive primitives_redesign checkout when building Lua tests")
    endif()
    add_executable(primitive_discovery_tests tests/primitive_discovery_tests.cpp)
    target_link_libraries(primitive_discovery_tests PRIVATE espressio_lua)
    target_include_directories(primitive_discovery_tests PRIVATE "${ESPRESSIO_PRIMITIVE_DIR}/src")
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(primitive_discovery_tests PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME primitive_discovery_tests COMMAND primitive_discovery_tests)

    set_tests_properties(binding_tests demo_test primitive_discovery_tests PROPERTIES TIMEOUT 30)
endif()
